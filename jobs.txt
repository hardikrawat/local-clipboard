<?php

namespace App\Jobs;

use App\Models\Order\Order;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;

abstract class BaseWorkflowJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public function __construct() {}

    abstract public function handle(): void;

    public function dispatchNextJob($payload): void
    {
        $order = Order::where('adid', $payload["orderId"])->first();
        if (!$order) {
            var_dump("Order not found for Adid: " . $payload["orderId"]);
            return;
        }

        $workflowConfig = $order
            ->productDefinition
            ->workflow_config;

        if ($workflowConfig && array_key_exists(static::class, $workflowConfig)) {
            $jobConfig = $workflowConfig[static::class];
            if ($jobConfig && isset($jobConfig['nextJob']) && $jobConfig['nextJob']) {
                $jobConfig['nextJob']::dispatch($payload);
            }
        }
    }
}

/**
 * Data Integrity Check job handler.
 */
class DataIntegrityCheckJobV4 extends BaseWorkflowJob implements ShouldQueue
{
    protected $payload;

    protected $sourceSystemId;

    protected $orderId;

    protected OrderRepository $orderRepository;

    protected EventService $eventService;

    protected ReportRepository $reportRepository;

    const DEFAULT_EVENT_CONTEXT = 'ORDER_CHECK_DATA_INTEGRITY';

    const EVENT_UDID = 'WIP_INTCHKP_MPDD_DI';

    const JOB_VERSION = 4;

    /**
     * Create a new job instance.
     */
    public function __construct($payload)
    {
        $this->payload = $payload;
        $this->orderRepository = app(OrderRepository::class);
        $this->eventService = app(EventService::class);
        $this->reportRepository = app(ReportRepository::class);
    }

    public function handle(): void
    {
        try {
            DB::beginTransaction();
            $this->orderId = $this->payload['orderId'];

            $inboundApiRequestId = $this->payload['apiRequestId'];
            $jobId = (int) $this->job->getJobId();
            $testCode = $this->payload['orderedTest'];
            
            $order = $this->orderRepository->getOrder($this->orderId);
            $this->sourceSystemId = $this->orderRepository->getOrderSourceSystemId($this->orderId);

            $sourceSystem = SourceSystem::find($this->sourceSystemId);

            $failureReasons = [];

            $event = $this->eventService->storeOrderEvent(
                $order,
                self::EVENT_UDID,
                $sourceSystem->udid,
                $inboundApiRequestId,
                $jobId,
                0,
                $testCode
            );

            $checkPartialReportIsGenerated = $this->reportRepository->checkReportStatus($this->orderId, 'PRELIM');
            if (!$checkPartialReportIsGenerated) {
                $failureReasons[] = 'Partial report is not generated yet for the linked profile order.';
            }

            if(!empty($failureReasons)){
                $this->eventService->setConditioCode($order, $event->order_event_id, 'MDICFL');
                $this->eventService->setConditioCode($order, $event->order_event_id, 'WIPH');

                $this->orderRepository->addCommentsToEvents([
                    'order_event_id' => $event->order_event_id,
                    'systemComment' => 'Data Integrity Check failed: '.implode(', ', $failureReasons),
                    'systemCommentTime' => Carbon::now(),

                ]);
            } else {
                $this->eventService->setConditioCode($order, $event->order_event_id, 'MDICOK');
                $orderJobSpawner = app(OrderJobSpawnerV4::class);
                $orderJobSpawner->spawn(get_class($this), [
                    'orderId' => $this->orderId,
                    'data' => [
                       'orderId' => $this->orderId,
                       'apiRequestId' => $inboundApiRequestId,
                       'eventUdid' => self::EVENT_UDID,
                       'reportAmendedWithTNPT' => false,
                       'correctedValues' => [],
                       'rptStatusUdid' => null,
                       'orderedTest' => $testCode,
                       'conditionCode' => $this->payload['conditionCode'],
                       'data' => $this->payload,
                    ],
                ]);
            }
            DB::commit();
        } catch (Throwable $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }
            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT
            );

            $this->fail($customException);
        }
    }
}

class Hl7DeliveryJobV4 implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    protected $payload;

    protected SystemInterfaceModel $interfaceModel;

    protected Order $order;

    protected OrderService $orderService;

    protected EventService $eventService;

    protected OrderResultRepository $orderRepository;

    protected QWorkerOrder $qWorkerOrder;

    protected SourceSystemModel $systemModel;

    protected JobModel $jobModel;

    protected OrderConditionRepository $orderConditionRepository;

    const DEFAULT_EVENT_CONTEXT = 'ORDER_RESULT_PUBLISH';

    const HL7_INTERFACE_UDID_ID = 'UNE_ORU_M8Q_HL7v2.3_v2';

    public function __construct($payload = [])
    {
        $this->payload = $payload;
        $this->qWorkerOrder = app(QWorkerOrder::class);
        $this->orderService = app(OrderService::class);
        $this->eventService = app(EventService::class);
        $this->interfaceModel = app(SystemInterfaceModel::class);
        $this->jobModel = app(JobModel::class);
        $this->orderConditionRepository = app(OrderConditionRepository::class);
        $this->orderRepository = app(OrderResultRepository::class);
        $this->systemModel = app(SourceSystemModel::class);
    }

    public function handle()
    {
        try {
            DB::beginTransaction();
            $orderId = $this->job->getOrderId();
            $this->order = $this->orderService->getOrderById($orderId);

            $eventUdid = 'OUT_ORD_RESUPLD';
            $inboundApiRequestId = $this->payload['apiRequestId'];
            $testCodes = $this->payload['test_code'];

            $uri = $this->qWorkerOrder->getInterfaceEndpoint(self::HL7_INTERFACE_UDID_ID, $eventUdid);

            $apiPayload = $this->getApiPayload($this->order, $this->payload['report_id']);
            $request = app(Request::class);
            $authorization = $request->getServiceAuthorization(self::HL7_INTERFACE_UDID_ID);

            $response = $request->callOutboundApiRequest(
                $uri,
                [
                    'Authorization: '.$authorization['token'],
                    'Client-Id: 1',
                ],
                $apiPayload,
                $authorization['systemUserApiCredentialId']
            );

            if (! in_array($response['responseStatusCode'], [200, 201])) {
                throw new Exception('Order transmission to Mulesoft failed');
            }
            $this->jobModel->addOutboundRequests($this->job->getLogId(), [$response['apiRequestId']]);

            $orderEvent = $this->eventService->storeOrderEvent(
                $this->order,
                $eventUdid,
                $this->payload['sourceSystemUdid'],
                $inboundApiRequestId,
                $this->job->getJobId(),
                $response['apiRequestId'],
                $testCodes
            );

            $status = $this->payload['rptStatusUdid'];

            if ($status === 'CORRECTED') {
                $conditionCode = 'RTRANC';
            } elseif ($status === 'FINAL') {
                $conditionCode = 'RTRANF';
            } else {
                $conditionCode = 'RTRANP';
            }
            $conditionDetail = $this->orderService->getCondition($conditionCode);
            $this->order->conditions()->create([
                'gid_condition_id' => $conditionDetail->gid_condition_id,
            ]);
            $this->createOrderConditionHistory($this->order, $orderEvent, $conditionDetail->gid_condition_id);
            DB::commit();
        } catch (Throwable $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT,
                null
            );
            $this->fail($customException);
        }
    }

    protected function getApiPayload(Order $order, int $reportId): array
    {

        $report = $order->report()->where('report_id', $reportId)->first();
        $reportdetail = $report->reportDetails()->where('report_id', $report->report_id)->where('type', 'HL72.3')->first();

        $interfaceQdibParams = json_decode($this->interfaceModel->getInterfaceMeta(self::HL7_INTERFACE_UDID_ID), true)['qdibParams'];
        $gtwQdibParams = json_decode($this->systemModel->getSourceSystemMeta('GTW'), true)['qdibParams'];

        $srcId = $interfaceQdibParams['qdibSrcInfXref'].$gtwQdibParams['qdibDefaultDelim'].
            $gtwQdibParams['qlsSsidXref'].$gtwQdibParams['qdibDefaultDelim'].$gtwQdibParams['qdibDefaultSuff'];

        if (empty($order->detail['provider']['physicianRequisitionId'])) {
            $reqNum = 'NA';
        } else {
            $reqNum = $order->detail['provider']['physicianRequisitionId'];
        }

        $payload = [
            'QdibEnvelope' => [
                'header' => [
                    'srcId' => $srcId,
                    'appTs' => Carbon::now()->format('Y-m-d\TH:i:s'),
                    'log' => $interfaceQdibParams['qdibLog'],
                    'destId' => $interfaceQdibParams['qdibDestQueue'],
                    'type' => $interfaceQdibParams['qdibMessageType'],
                ],
                'body' => [
                    'rec' => [
                        'md' => [
                            'key' => [
                                ['nm' => 'clientNum', 'vl' => $order->detail['client']['clientId']],
                                ['nm' => 'seqNum', 'vl' => $report->sequence],
                                ['nm' => 'reqNum', 'vl' => $reqNum],
                                ['nm' => 'accNum', 'vl' => explode('.', $order->adid)[2]],
                            ],
                        ],
                        'pl' => base64_encode($reportdetail->value),
                    ],
                ],
            ],
        ];

        return $payload;
    }

    protected function createOrderConditionHistory(Order $order, OrderEvent $orderEvent, int $conditionCodeId): void
    {
        $order->conditionHistory()->create([
            'gid_condition_id' => $conditionCodeId,
            'order_event_id' => $orderEvent->order_event_id,
        ]);
    }
}

namespace App\Jobs;

use App\Exceptions\CustomJobException;
use App\Jobs\Harvester\OrderMessageHarvester;
use App\Jobs\Harvester\WorkflowTrait;
use App\Jobs\Parser\OrderMessageParser;
use App\Models\Order\Order;
use App\QWorker\JobModel\ApiModel;
use App\QWorker\Traits\OrderJobHelperTrait;
use App\Repositories\Queue\QueueRepository;
use DB;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;
use Throwable;

class OrderEventDiscoveryJobV4 implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, OrderJobHelperTrait, Queueable, SerializesModels, WorkflowTrait;

    const DEFAULT_EVENT_CONTEXT = 'ORDER_EVENT_DISCOVERY';

    /**
     * for post-mpdd job implementations
     */
    const JOB_VERSION = 4;

    public function __construct(
        protected array $payload,
    ) {}

    public function handle(): void
    {
        try {
            DB::beginTransaction();
            $payload = $this->payload;

            $api = app(ApiModel::class);

            $apiPayload = $this->getMessage($api, $payload['apiRequestId']);
            $message = $apiPayload['messages'][$payload['messageIndex']];
            $payload['interfaceUdid'] = $apiPayload['interfaceUdid'] ?? $payload['interfaceUdid'];
            $payload['messages'] = $apiPayload['messages'];
            $parser = app(OrderMessageParser::class);
            $harvester = app(OrderMessageHarvester::class);

            $parser->setMessage($this->getPayload($message))->parse();
            $harvester->setParser($parser)->setInterfaceUdid($apiPayload['interfaceUdid'])->harvest();
            $orderAdid = $harvester->getAdid();
            $order = Order::where('adid', $orderAdid)->first();
            if (! empty($order?->order_id)) {
                $payload['orderId'] = $order->order_id;
            }
            $nextJob = $harvester->getNextJob();
            $eventMeta = $harvester->getEventMeta();

            if (! class_exists($nextJob) || ! method_exists($nextJob, 'dispatch')) {
                throw new \RuntimeException("Invalid job class: {$nextJob}");
            }

            $orderInductionJobPayload = [
                'job' => $nextJob,
                'job_type' => $eventMeta['nextJobQueue'],
                'data' => $payload,
            ];

            $arr = [
                'queue' => $orderInductionJobPayload['job_type'],
                'payload' => new $orderInductionJobPayload['job']($payload),
            ];

            $fullArr = array_merge($arr, $payload);
            app(QueueRepository::class)->queue(
                'Illuminate\Queue\CallQueuedHandler@call',
                $fullArr,
                self::JOB_VERSION
            );

            DB::commit();
        } catch (Throwable $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            // Use the CustomJobException
            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT
            );

            $this->fail($customException);
        }
    }
}

namespace App\Jobs;

use App\Exceptions\CustomJobException;
use App\Http\Services\V4\EventService;

class OrderEventUpdateJobV4 extends BaseWorkflowJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    /**
     * @var array Incoming job payload
     */
    protected array $payload;

    /**
     * @var int|null Internal event history tracking (not used directly)
     */
    protected ?int $eventHistoryId = null;

    /**
     * @var OrderRepository Handles database interaction with order conditions
     */
    protected OrderRepository $orderRepository;

    const STARLIMS_INTERFACE_UDID = 'STARLIMS_HAYSTACK_ORDERS';

    const DEFAULT_EVENT_CONTEXT = 'ORDER_EVENT_CONSUME';

    const JOB_VERSION = 4;

    /**
     * Constructor
     *
     * @param  array  $payload  The job payload containing order and processing info
     * @param  OrderRepository  $orderRepository  Repository for accessing order conditions
     */
    public function __construct(array $payload)
    {
        $this->payload = $payload;
        $this->orderRepository = app(OrderRepository::class);
    }

    /**
     * Main job handler
     *
     * Processes the incoming order update event by:
     * - Updating patient/client/test data
     * - Creating order events
     * - Setting/unsetting condition codes
     */
    public function handle(): void
    {
        try {
            DB::beginTransaction();

            $orderId = $this->job->getOrderId();
            $jobId = $this->job->getLogId();
            $order = $this->orderRepository->getOrder($orderId);
            $productDefinition = $order->productDefinition;

            $updatedResultValues = $this->payload['testData'] ?? [];
            if (! empty($this->payload['newPatientData']) || isset($this->payload['newClientData']) || ! empty($updatedResultValues)) {
                if (! empty($order)) {
                    $orderDetails = $order->detail;

                    if (! empty($this->payload['newPatientData'])) {
                        $this->updateOrderPatient($this->payload['newPatientData'], $orderDetails);
                    }

                    if (isset($this->payload['newClientData'])) {
                        $this->updateClient($this->payload['newClientData'], $orderDetails);
                    }

                    if (! empty($updatedResultValues)) {
                        $this->updateOrderTestData($updatedResultValues, $orderId);
                    }
                    $order->detail = $orderDetails;
                    $order->save();
                    $eventService = app(EventService::class);
                    $testCodes = $this->orderRepository->getTestCode($order);
                    $testCodes = ! empty($testCodes) ? explode(',', $testCodes)[0] : '';
                    $eventService->storeOrderEvent(
                        $order,
                        $this->payload['systemEventUdid'],
                        $order->detail['meta']['sourceSystemUdid'],
                        $this->payload['apiRequestId'],
                        $jobId,
                        0,
                        $testCodes
                    );
                }

                $orderedTest = $productDefinition->test_code;

                if ($productDefinition && $productDefinition->config_data) {
                    $endpointRouting = $productDefinition->config_data;
                    $publishConfig = $endpointRouting['PublishToTechnical'] ?? null;
                    if ($publishConfig) {
                        $newPayload = [
                            'orderId' => $orderId,
                            'event' => $this->payload['systemEventUdid'],
                            'apiRequestId' => $this->payload['apiRequestId'],
                            'jobChainOriginSourceSystemId' => $this->payload['jobChainOriginSourceSystemId'],
                            'interfaceUdid' => $publishConfig['interfaceUdid'] ?? self::STARLIMS_INTERFACE_UDID,
                            'eventUdid' => $publishConfig['eventUdid'] ?? 'WIP_OUT_ORDER_TECH',
                            'newPatientData' => $this->payload['newPatientData'],
                            'newClientData' => $this->payload['newClientData'],
                            'existingPatientAndClientData' => $this->payload['existingPatientAndClientData'],
                            'outboundInterfaceExpectedResponse' => $publishConfig['outboundInterfaceExpectedResponse'] ?? '',
                            'testXref' => $orderedTest,
                            'updatedResultValues' => $updatedResultValues,
                        ];

                        $orderJobSpawner = app(OrderJobSpawnerV4::class);
                        $orderJobSpawner->spawn(get_class($this), [
                            'orderId' => $orderId,
                            'data' => $newPayload,
                        ]);
                    }
                }
            }
            DB::commit();
        } catch (Exception $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT,
                null
            );
            $this->fail($customException);
        }
    }

    /**
     * Update patient fields inside the order details
     *
     * @param  array  $testData  The updated patient info
     * @param  array  &$orderDetails  Reference to the full order detail array
     */
    protected function updateOrderPatient(array $newPatientData, array &$orderDetails): void
    {
        foreach ($newPatientData as $sourceKey => $targetFieldValue) {
            if (isset($orderDetails['patient'][$sourceKey])) {
                if (! is_null($targetFieldValue)) {
                    $orderDetails['patient'][$sourceKey] = $targetFieldValue;
                }
            }
        }
    }

    /**
     * Update client-related data in order details
     *
     * @param  array  $testData  The new client values
     * @param  array  &$orderDetails  Reference to order detail structure
     */
    protected function updateClient(array $newClientData, array &$orderDetails): void
    {
        foreach ($newClientData as $sourceKey => $targetFieldValue) {
            if (isset($orderDetails['client'][$sourceKey])) {
                if (! is_null($targetFieldValue)) {
                    $orderDetails['client'][$sourceKey] = $targetFieldValue;
                }
            }
        }
    }

    /**
     * Update test data for a specific order
     *
     * @param  array  $updatedTestData  New test result data
     * @param  int  $orderId  The order's unique ID
     */
    protected function updateOrderTestData(array $updatedTestData, int $orderId): void
    {
        foreach ($updatedTestData as $testId => $groupedRecords) {
            $test = Test::where('order_id', $orderId)
                ->where('test_id', $testId)
                ->first();

            if (! $test) {
                continue;
            }
            $existingDetail = $test->detail ?? [];
            foreach ($groupedRecords as $groupIndex => $updatedRecords) {
                $indexNumber = $this->findIndexByKey($existingDetail[0], 'SRCId', $updatedRecords['SRCId']);
                if (! is_null($indexNumber)) {
                    $existingDetail[0][$indexNumber] = $updatedRecords;
                }
            }
            $test->detail = $existingDetail;
            $test->save();
        }
    }

    public function findIndexByKey(array $array, string $targetKey, string $targetValue)
    {
        foreach ($array as $index => $element) {
            if (is_array($element) && isset($element[$targetKey]) && $element[$targetKey] === $targetValue) {
                return $index;
            }
        }

        return null;
    }
}

class OrderEventUpdatePrepJobV4 extends BaseWorkflowJob implements ShouldQueue
{

    private array $payload;

    const DEFAULT_EVENT_CONTEXT = 'ORDER_EVENT_CONSUME';

    const JOB_VERSION = 4;

    protected $orderModel;

    public function __construct(array $payload)
    {
        $this->payload = $payload;
        $this->orderModel = app(OrderRepository::class);
    }

    /**
     * Execute the job.
     */
    public function handle(): void
    {
        try {
            DB::beginTransaction();

            $parser = app(OrderMessageParser::class);
            $harvester = app(OrderMessageHarvester::class);

            $orderId = $this->job->getOrderId();

                ->setParser($parser)

            $eventUdid = $harvestedData['meta']['eventUdid'];

            $newPayload['existingPatientAndClientData'] = $patientAndClientData;
            $newPayload['newPatientData'] = $this->getChangedPatientData($harvestedData['fields'], $patientAndClientData);
            $newPayload['newClientData'] = $this->getChangedClientData($harvestedData['fields'], $patientAndClientData);

            $testData = $this->getMultiTestDataFormat($harvestedData, $order);
            $newPayload['apiRequestId'] = $this->payload['apiRequestId'];
            $newPayload['systemEventUdid'] = $eventUdid;
            $newPayload['testData'] = $testData;
            $newPayload['orderId'] = $orderId;
            $newPayload['jobChainOriginSourceSystemId'] = $this->getSourceSystemId($harvestedData['meta']['sourceSystemUdid']);

            $orderInductionJobPayload = [
                'job' => OrderEventUpdateJobV4::class,
                'job_type' => 'UPDATE',
                'data' => $newPayload,
            ];
            $arr = [
                'queue' => $orderInductionJobPayload['job_type'],
                'payload' => new $orderInductionJobPayload['job']($orderInductionJobPayload['data']),
            ];
            $fullArr = array_merge($arr, $orderInductionJobPayload['data']);
            app(QueueRepository::class)->queue(
                'Illuminate\Queue\CallQueuedHandler@call',
                $fullArr,
                self::JOB_VERSION
            );
            DB::commit();
        } catch (Exception $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT,
                null
            );
            $this->fail($customException);
        }
    }

    private function getMultiTestDataFormat(array $harvestedData, Order $order): array
    {
        $final = [];
        foreach ($order->tests as $test) {
            $productDefinitonData = ProductDefinitions::where('test_code', $test->test_code)
                ->first();
            $obxData = $productDefinitonData->test_structure['properties']['items']['elements']['obx']['elements'] ?? [];

            $testId = $test->test_id;
            if (empty($test->detail)) {
                continue;
            }
            $existingDataMap = [];
            foreach ($test->detail as $groupIndex => $group) {
                foreach ($group as $obx) {
                    if (! empty($obx['SRCId'])) {
                        $existingDataMap[$obx['SRCId']] = $obx;
                    }
                }
            }

            foreach ($harvestedData['items'] as $itemIndex => $item) {
                if (empty($item['obx'])) {
                    continue;
                }

                $updatedObxList = [];
                foreach ($item['obx'] as $obx) {
                    $srcId = $obx['SRCId'] ?? null;
                    if (! $srcId) {
                        continue;
                    }

                    if (! isset($existingDataMap[$srcId])) {
                        $updatedObxList[] = $this->updateNomenclature($obx, $obxData);
                    } elseif ($existingDataMap[$srcId] != $obx) {
                        $updatedObxList[] = $this->updateNomenclature($obx, $obxData);
                    }
                }

                if (! empty($updatedObxList)) {
                    $final[$testId] = $updatedObxList;
                }
            }
        }

        return $final;
    }

    public function updateNomenclature(array $hl7Data, array $obxData): array
    {
        foreach ($obxData as $key => $value) {
            if ($hl7Data['99QDISRCID'] == $value['src']) {
                $hl7Data['nomenclature'] = $value['nomenclature'];
            }
        }

        return $hl7Data;
    }

    public function getSourceSystemId(string $sourceSystemUdId): int
    {
        $sourceSystem = SourceSystem::where('udid', $sourceSystemUdId)->first();
        if (! isset($sourceSystem->source_system_id)) {
            throw new \RuntimeException("SourceSystem not found for UDID: {$sourceSystemUdId}");
        }

        return $sourceSystem->source_system_id;
    }

    public function getOrderDetails(int $orderId): array
    {
        $order = Order::where('order_id', $orderId)->first();

        return $order->detail;
    }

    protected function getChangedPatientData(array $incomingData, array $existingData): array
    {
        $existingData['patient']['patientDOB'] = str_replace('-', '', $existingData['patient']['patientDOB']);
        $updatedValues = [];
        foreach ($incomingData as $key => $value) {
            if (isset($existingData['patient'][$key])) {
                if (isset($value) && ($existingData['patient'][$key] != $value)) {
                    $updatedValues[$key] = $value;
                }
                if ($existingData['patient'][$key] === 'patientInitials' && empty($value)) {
                    $updatedValues['patientInitials'] = '';
                }
            }
        }

        return $updatedValues;
    }

    protected function getChangedClientData(array $incomingData, array $existingData): array
    {
        $updatedValues = [];

        $updatedValues['xref'] = $existingData['client']['clientId'];

        foreach ($incomingData as $key => $value) {
            if (isset($existingData['client'][$key])) {
                if (isset($value) && $existingData['client'][$key] != $value) {
                    $updatedValues[$key] = $value;
                }
            }
        }

        return $updatedValues;
    }
}

     */

        protected array $payload,

        $this->orderRepository = app(OrderRepository::class);
    }

    /**
     * Execute the job.
     */
    public function handle(): void
    {
        try {
            DB::beginTransaction();
            $payload = $this->payload;
            $api = app(ApiModel::class);

            $apiPayload = $this->getMessage($api, $payload['apiRequestId']);

            $message = $apiPayload['messages'][$payload['messageIndex']];

            $this->harvester = app(OrderMessageHarvester::class);
            $parser = app(OrderMessageParser::class);
            $parser->setMessage($this->getPayload($message))->parse();

            $inboundApiRequestId = $payload['apiRequestId'];
            $interfaceUdid = $payload['interfaceUdid'];

            $harvestedData = $this
                ->harvester
                ->setParser($parser)
                ->setInterfaceUdid($interfaceUdid)
                ->harvest();

            $adid = $this->harvester->getAdid();
            // preserve a copy of original harvested data for validation
            $originalHarvestedData = json_decode(json_encode($harvestedData), true);  // Deep copy

            $harvestedData['patient'] = $this->getPatient($harvestedData);
            $harvestedData['provider'] = $this->getProvider($harvestedData);
            $harvestedData['client'] = $this->getClient($harvestedData);

            $combined = array_merge($harvestedData['patient'], $harvestedData['provider'], $harvestedData['client']);
            $uniqueFields = array_diff_key($harvestedData['fields'], $combined);
            $harvestedData['fields'] = $uniqueFields;

            // Analyze all items to determine order structure
            $orderStructure = $this->analyzeOrderStructure($harvestedData['items'], $originalHarvestedData, $interfaceUdid);

            $harvestedData['items'] = $orderStructure['updatedItems'];
            $order = $this->createOrder($orderStructure, $harvestedData, $adid);
            $jobModel = app(JobModel::class);
            $jobId = isset($this->payload['job_id']) ? (int) $this->payload['job_id'] : (int) $this->job->getLogId();
            $ordeId = isset($payload['orderId']) ? $payload['orderId'] : (int) $order->order_id;

            $jobModel->setJobOrderId($jobId, $ordeId);

            $this->createAllTests($order, $orderStructure, $harvestedData);

            // TODO check if we should add $CompleteJobEvent->check as OrderInductionJobV2

            if ($order) {
                $this->createNewOrderCondition($order, $inboundApiRequestId);
                $orderJobSpawner = app(OrderJobSpawnerV4::class);

                $orderJobSpawner->spawn(get_class($this), [
                    'orderId' => $ordeId,
                    'data' => [
                        'apiRequestId' => $payload['apiRequestId'],
                        'address' => $parser->getAddress(),
                    ],
                ]);
            }

            DB::commit();
        } catch (Throwable $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }
            // Use the CustomJobException
            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT
            );

            $this->fail($customException);
        }
    }

    private function transformObxSegments(array $harvestedItems, array $obxesInProductDefinition): array
    {
        $updatedObxes = [];
        foreach ($harvestedItems as $obxSegment) {
            foreach ($obxesInProductDefinition['elements'] as $key => $structure) {
                if ($obxSegment['SRCId'] === $structure['src']) {
                    $updatedObxes[$key] = $obxSegment;

                    continue;
                }
            }
        }

        return $updatedObxes;
    }

    private function analyzeOrderStructure(array $items, array $originalHarvestedData, string $interfaceUdid): array
    {
        $parentTestCodes = [];
        $productDefinitions = [];
        $validatedTests = [];
        $updatedItems = [];

        // First pass: collect all parent test codes and validate
        foreach ($items as $index => $harvestedItem) {
            $testCodeXref = str_replace('%', '', $harvestedItem['fields']['99QDITestID']);
            $prodDefinition = ProductDefinitions::where([
                'test_code_xref' => $testCodeXref,
                'is_active' => 1,
            ])->first();

            $this->validateInterfaceUdid($interfaceUdid, $prodDefinition->product_definitions_id, $prodDefinition->test_code);
            $this->validateOrderInput($originalHarvestedData, $prodDefinition);

            $productDefinitions[$index] = $prodDefinition;
            $validatedTests[] = $index;

            $updatedItem = $harvestedItem;
            if (isset($harvestedItem['obx'])) {
                $obxesStructure = $prodDefinition->test_structure['properties']['items']['elements']['obx'];
                $transformedObxes = $this->transformObxSegments($harvestedItem['obx'], $obxesStructure);
                $updatedItem['obx'] = $transformedObxes;
            }
            $updatedItems[$index] = $updatedItem;

            if ($prodDefinition->parent_test_code) {
                $parentTestCodes[] = $prodDefinition->parent_test_code;
            }
        }

        // Determine order structure based on parent test codes
        $uniqueParentCodes = array_unique($parentTestCodes);
        $orderType = 'standalone';  // Default
        $orderProductDefinition = null;
        $parentTestCode = null;

        if (count($uniqueParentCodes) === 1) {
            // All tests share the same parent - use parent for order
            $parentTestCode = $uniqueParentCodes[0];
            $parentProdDef = ProductDefinitions::where([
                'test_code' => $parentTestCode,
                'is_active' => 1,
            ])->first();

            $this->validateInterfaceUdid($interfaceUdid, $parentProdDef->product_definitions_id, $parentTestCode);
            $this->validateOrderInput($originalHarvestedData, $parentProdDef);

            $orderType = 'parent';
            $orderProductDefinition = $parentProdDef;
        }

        // If no valid parent structure, use the first validated test for the order
        if (! $orderProductDefinition && ! empty($validatedTests)) {
            $firstTestIndex = $validatedTests[0];
            $orderProductDefinition = $productDefinitions[$firstTestIndex];
            $orderType = 'standalone';
        }

        return [
            'type' => $orderType,
            'productDefinition' => $orderProductDefinition,
            'parentTestCode' => $parentTestCode,
            'testDefinitions' => $productDefinitions,
            'validatedTests' => $validatedTests,
            'updatedItems' => $updatedItems,
        ];
    }

    private function createOrder(array $orderStructure, array $harvestedData, string $adid): Order
    {
        if (! $orderStructure['productDefinition']) {
            throw new RuntimeException('No valid product definition found for order creation');
        }

        $order = Order::create([
            'product_definitions_id' => $orderStructure['productDefinition']->product_definitions_id,
            'detail' => array_merge(
                $this->removeObxSections($harvestedData),
                $harvestedData['patient'],
                $harvestedData['provider'],
                $harvestedData['client']
            ),
            'adid' => $adid,
        ]);

        return $order;
    }

    private function createAllTests(Order $order, array $orderStructure, array $harvestedData): void
    {
        $parentTest = null;

        // If this is a parent-type order, create the parent test first
        if ($orderStructure['type'] === 'parent' && $orderStructure['parentTestCode']) {
            $parentTest = $this->createParentTest($order, $orderStructure['parentTestCode'], $harvestedData);
        }

        if ($parentTest) {
            $this->createSamples($parentTest, $harvestedData);
        }

        // Create all child/regular tests
        foreach ($orderStructure['validatedTests'] as $itemIndex) {
            $harvestedItem = $harvestedData['items'][$itemIndex];
            $testDefinition = $orderStructure['testDefinitions'][$itemIndex];

            // Determine if this should be a child test
            $shouldBeChild = ($orderStructure['type'] === 'parent' &&
                $testDefinition->parent_test_code === $orderStructure['parentTestCode']);

            $singleTest = $this->createTest($order, $harvestedItem, $shouldBeChild ? $parentTest : null);
            if (! $parentTest) {
                $this->createSamples($singleTest, $harvestedData);
            }
        }
    }

    private function createParentTest(Order $order, string $parentTestCode, array $harvestedData): ?Test
    {
        $parentTestCodeProdDef = ProductDefinitions::where([
            'test_code' => $parentTestCode,
            'is_active' => 1,
        ])->first();

        $parentTest = null;
        // TODO determine from where a non-virtual parent test would get its detail
        if ($parentTestCodeProdDef->is_virtual) {
            $parentTest = $order->tests()->create([
                'test_code' => $parentTestCode,
                'test_code_xref' => $parentTestCode,
                'product_definitions_id' => $parentTestCodeProdDef->product_definitions_id,
                'detail' => [],  // Virtual tests have empty detail
                'parent_id' => null,
            ]);
        }

        return $parentTest;
    }

    private function validateOrderInput($harvestedData, $productDefinitions)
    {
        $testStructure = $productDefinitions->test_structure;
        $validator = new OrderValidator($testStructure);
        $errors = $validator->validate($harvestedData);

        if (! empty($errors)) {
            Log::error('JSON validation errors:', $errors);
            throw new RuntimeException('Failed to validate order data: '
                .$productDefinitions->test_code.' - '.json_encode($errors, JSON_PARTIAL_OUTPUT_ON_ERROR));
        }
    }

    private function getProductDefinition(string $testCode): ProductDefinitions
    {
        return ProductDefinitions::where([
            'test_code' => $testCode,
            'is_active' => 1,
        ])->firstOr(function () use ($testCode) {
            Log::error('Failed to fetch product definitions', [
                'test_code' => $testCode,
            ]);
            throw new RuntimeException('Failed to fetch product definitions for test code: '.$testCode);
        });
    }

    private function validateInterfaceUdid(string $interfaceUdid, int $productDefinitionsId, string $testCode): void
    {
        $interfaceIdList = ProductInterfaceConfiguration::where('product_definitions_id', $productDefinitionsId)
            ->pluck('interface_configuration_id')
            ->toArray();

        $interfaceId = InterfaceConfiguration::where('interface_udid', $interfaceUdid)
            ->value('interface_configuration_id');

        // Check if the provided interface UDID exists in the list
        if (! in_array($interfaceId, $interfaceIdList)) {
            Log::error('Interface UDID not found in product interface configurations.', [
                'interfaceUdid' => $interfaceUdid,
                'productDefinitionsId' => $productDefinitionsId,
            ]);

            throw new RuntimeException('Invalid interface UDID: '.$interfaceUdid.' for test code: '.$testCode);
        }
    }

    private function createTest(Order $order, array $harvestedItem, ?Model $parentTest = null): Test
    {
        $testCodeXref = $harvestedItem['fields']['99QDITestID'];
        $testCodeXref = str_replace('%', '', $testCodeXref);
        $testCodeProdDef = ProductDefinitions::where([
            'test_code_xref' => $testCodeXref,
            'is_active' => 1,
        ])->first();

        if ($parentTest) {
            $this->verifyParentTestCode($testCodeProdDef, $parentTest->test_code);
        }
        $obxData = $testCodeProdDef->test_structure['properties']['items']['elements']['obx']['elements'] ?? [];

        $testCode = $testCodeProdDef->test_code;

        return $order->tests()->create([
            'test_code' => $testCode,
            'product_definitions_id' => $testCodeProdDef->product_definitions_id,
            'test_code_xref' => $testCodeXref,
            'product_definitions_id' => $testCodeProdDef->product_definitions_id,
            'detail' => $this->getObxSections($harvestedItem, $obxData),
            'parent_id' => $parentTest?->test_id,
        ]);
    }

    private function verifyParentTestCode($testCodeProductDefinition, $parentTestCode): void
    {
        if ($testCodeProductDefinition->parent_test_code !== $parentTestCode) {
            Log::error('Parent test code mismatch', [
                'expected' => $testCodeProductDefinition->parent_test_code,
                'actual' => $parentTestCode,
            ]);
            throw new RuntimeException('Parent test code mismatch for test code: '.$testCodeProductDefinition->test_code
                .'. Expected: '.$testCodeProductDefinition->parent_test_code
                .', Actual: '.$parentTestCode);
        }
    }

    // we remove the obx section from order detail, since it goes to test info
    private function removeObxSections(array $data): array
    {
        foreach ($data as $key => $value) {
            if (is_array($value)) {
                if ($key === 'obx') {
                    unset($data[$key]);
                } else {
                    $data[$key] = $this->removeObxSections($value);
                }
            }
        }

        return $data;
    }

    private function getObxSections(array $data, array $prodOBXData): array
    {
        $obxSections = [];

        foreach ($data as $key => $value) {
            if (is_array($value)) {
                if ($key === 'obx') {
                    $obxSections[] = $this->updateNomenclature($value, $prodOBXData);
                } else {
                    $obxSections = array_merge($obxSections, $this->getObxSections($value, $prodOBXData));
                }
            }
        }

        return $obxSections;
    }

    public function updateNomenclature(array $hl7Data, array $obxData): array
    {
        foreach ($hl7Data as $key => $value) {
            if (array_key_exists($key, $obxData)) {
                $hl7Data[$key]['nomenclature'] = $obxData[$key]['nomenclature'];
            }
        }

        return $hl7Data;
    }

    private function createNewOrderCondition(Order $order, $inboundApiRequestId): void
    {
        $openConditionCodeId = $this->getConditionId('OPEN');

        $sourceSystemUdid = $this->harvester->getSourceSystemUdid();

        $this->createCondition($order, $openConditionCodeId);
        $eventService = app(EventService::class);
        $jobId = isset($this->payload['job_id']) ? (int) $this->payload['job_id'] : (int) $this->job->getLogId();
        $testCodes = $this->orderRepository->getTestCode($order);
        $testCodes = ! empty($testCodes) ? explode(',', $testCodes)[0] : '';
        $orderEvent = $eventService->storeOrderEvent(
            $order,
            self::NEW_ORDER_EVENT,
            $sourceSystemUdid,
            $inboundApiRequestId,
            $jobId,
            0,
            $testCodes
        );
        $this->createOrderConditionHistory($order, $orderEvent, $openConditionCodeId);
    }

    private function getConditionId(string $conditionCode): int
    {
        $masterGidCondition = MasterGidCondition::where('udid', $conditionCode)->get(['gid_condition_id', 'udid'])->first();

        return $masterGidCondition->gid_condition_id;
    }

    private function createCondition(Order $order, int $conditionCodeId): OrderCondition
    {
        return $order->conditions()->create([
            'gid_condition_id' => $conditionCodeId,
        ]);
    }

    private function createOrderConditionHistory(Order $order, OrderEvent $orderEvent, int $conditionCodeId): void
    {
        $order->conditionHistory()->create([
            'gid_condition_id' => $conditionCodeId,
            'order_event_id' => $orderEvent->order_event_id,
        ]);
    }

    private function createSamples(Test $test, array $harvestedData): void
    {
        $samples = $this->getSamples($harvestedData);
        foreach ($samples as $sample) {
            if (! $sample['sampleTypeId']) {
                Log::error('Sample type ID is missing for sample', [
                    'sample' => $sample,
                ]);

                continue;  // Skip this sample if no sample type ID is found
            }
            $test->samples()->create([
                'sample_type_id' => $sample['sampleTypeId'],
                'sourceSystemXref' => $sample['sourceSystemXref'],
                'collectionDatetime' => $this->formatCollectionDate($sample['collectionDate'], 'Y-m-d H:i:s'),
                'collectionDatetimeTz' => $sample['collectionDatetimeTz'],
            ]);
        }
    }

    private function formatCollectionDate(string $datetimeStr, string $format): string
    {
        $dt = Carbon::createFromFormat('YmdHisO', $datetimeStr);

        return $dt->format($format);
    }

    private function getPatient(array $data): array
    {
        $fields = $data['fields'];

        return [
            'mrid' => array_exists('PMRN', $fields, null),
            'qdpnum' => array_exists('QDPNUM', $fields, null),
            'patientSex' => $fields['patientSex'],
            'patientDOB' => $fields['patientDOB'],
            'patientAge' => ($fields['patientAge'] ?? 0),
            'patientGivenName' => $fields['patientGivenName'],
            'patientSurname' => $fields['patientSurname'],
            'patientInitials' => $fields['patientInitials'],
            'patientNameSuffix' => $fields['patientNameSuffix'],
            'patientNamePrefix' => $fields['patientNamePrefix'],
            'patientNameTypeCode' => $fields['patientNameTypeCode'],
            'patientNameDegree' => $fields['patientNameDegree'],
            'patientMaidenSurname' => $fields['patientMaidenSurname'],
            'patientMaidenGivenName' => $fields['patientMaidenGivenName'],
            'patientMaidenAdditionalGivenNames' => $fields['patientMaidenAdditionalGivenNames'],
            'patientAlternateId' => $fields['patientAlternateId'],
            // 'orderExternalPatientId' => $orderExternalPatientId,
            // 'qdpidId' => $qdpidId,
        ];
    }

    private function getProvider(array $data): array
    {
        // @TODO These OPRV* should be replaced with jsonUdids or removed
        // kept here for compatibility with OrderMessageStoreV2
        $fields = $data['fields'];
        $npi = ($fields['OPRV_NTC'] ?? '') == 'NPI' && ! empty($fields['providerID'])
            ? $fields['providerID']
            : 0;

        // repeater field NPI
        $npiOprv = $fields['OPRVNPI'] ?? null;

        // in case NPI is different from direct field and repeater field
        if ($npi !== 0 && $npi === $npiOprv) {
            throw new \Exception('NPI mismatch detected from OPRVNPI and OPRV_NTC fields.');
        }

        $npi = $npiOprv ?? $npi;
        $givenName = $fields['physicianGivenName'] ?? $fields['OPRVNPI_GNM'] ?? '';
        $surname = $fields['physicianSurname'] ?? $fields['OPRVNPI_SNM'] ?? '';
        $initials = $fields['physicianAdditionalNames'] ?? $fields['OPRVNPI_MNM'] ?? '';
        $upin = $fields['OPRVUPIN'] ?? null;

        return [
            'npi' => $npi,
            'physicianGivenName' => $givenName,
            'physicianSurname' => $surname,
            'physicianAdditionalNames' => $initials,
            'providerID' => $fields['providerID'],
            'upin' => $upin,
            'physicianRequisitionId' => $fields['physicianRequisitionId'],
            'physicianAddressType' => $fields['physicianAddressType'],
            'physicianAddressPostal' => $fields['physicianAddressPostal'],
            'physicianAddressRegion' => $fields['physicianAddressRegion'],
            'physicianAddressStreet' => $fields['physicianAddressStreet'],
            'physicianAddressCountry' => $fields['physicianAddressCountry'],
            'physicianAddressLocality' => $fields['physicianAddressLocality'],
            'physicianAdditionalGivenNames' => $fields['physicianAdditionalGivenNames'],
        ];
    }

    private function getClient(array $data): array
    {
        $fields = $data['fields'];

        return [
            'clientId' => $fields['clientId'],
            'clientName' => $fields['clientName'],
            'clientPhone' => $fields['clientPhone'],
            'clientAddressPostal' => $fields['clientAddressPostal'],
            'clientAddressRegion' => $fields['clientAddressRegion'],
            'clientAddressCountry' => $fields['clientAddressCountry'],
            'clientAddressLocality' => $fields['clientAddressLocality'],
            'clientAddressStreetLN1' => $fields['clientAddressStreetLN1'],
            'clientAddressStreetLN2' => $fields['clientAddressStreetLN2'],
        ];
    }

    private function getSamples(array $data): array
    {
        $sourceSystemUdid = $data['meta']['sourceSystemUdid'];
        $sourceSystem = SourceSystem::where('udid', $sourceSystemUdid)->first();

        $sourceSystemId = null;
        if ($sourceSystem) {
            $sourceSystemId = (int) $sourceSystem->source_system_id;
        }

        $samples = $processedSamples = [];
        foreach ($data['items'] as $itemIndex => $test) {
            if (! array_exists('zsi', $test)) {
                // This test (OBR) doesn't have a ZSI (sample) segment so we skip
                continue;
            }

            $first = $test['zsi'][array_key_first($test['zsi'])];
            if (Arr::exists($test, 'zsi.0.qlsSampleID') && in_array($first['qlsSampleID'], $processedSamples)) {
                // Duplicate ZSI on a different OBR, ignore
                continue;
            }

            $sampleType = SampleType::select('sample_type_id')
                ->where('source_system_id', $sourceSystemId)
                ->where('source_system_xref', $first['qlsSampleType'])
                ->first();

            $sampleTypeId = null;
            if ($sampleType) {
                $sampleTypeId = (int) $sampleType->sample_type_id;
            }

            $samples[] = [
                'sampleTypeId' => $sampleTypeId,
                'qlsSampleType' => $first['qlsSampleType'],
                'sourceSystemXref' => $first['qlsSampleID'],
                'collectionDate' => $test['fields']['specimenCollectedOn'],
                'collectionDatetime' => $test['fields']['specimenCollectedOn'],
                'collectionDatetimeTz' => $test['fields']['specimenCollectedOnTZ'],
            ];

            $processedSamples[] = $first['qlsSampleID'];
        }

        return $samples;
    }

namespace App\Jobs;

use App\Models\Order\Order;
use App\Models\Tran\Queue\JobLog;
use App\QWorker\JobModel\Order\OrderModel;
use App\QWorker\Queue\JobSpawner;
use App\QWorker\Support\Arr;
use App\Repositories\Queue\QueueRepository;
use Illuminate\Database\Eloquent\Casts\Json;
use Mini\Model\Model;
use Illuminate\Support\Facades\Log;

/**
 * A generic job spawner.
 */
class OrderJobSpawnerV4 extends JobSpawner
{
    /**
     * Job queue.
     *
     * @var \App\QWorker\JobModel\Order\OrderModel|null
     */
    protected $model = null;

    /**
     * Order data.
     *
     * @var \App\QWorker\Order\Order
     */
    protected $order = null;

    const JOB_VERSION = 4;

    /**
     * Setup.
     *
     * @param  Order  $order  order logic
     * @param  OrderModel  $model  order data
     */
    public function __construct(Order $order, OrderModel $model)
    {
        parent::__construct();

        $this->model = $model;
        $this->order = $order;
    }

    public function queueMessage(array $interfaceConfig, int $apiRequestId, int $messageIndex): JobLog|Model
    {
        // Prepare the payload
        $payload = [
            'apiRequestId' => $apiRequestId,
            'messageIndex' => $messageIndex,
            'dispatcherJobConfig' => $interfaceConfig['dispatcherJobConfig'],
        ];

        // Pass the payload when creating the job
        return app(QueueRepository::class)->queue(
            'Illuminate\\Queue\\CallQueuedHandler@call',
            [
                'queue' => $interfaceConfig['dispatcherJobToQueue']['queueType'],
                'payload' => new $interfaceConfig['dispatcherJobToQueue']['job']($payload), // Pass the payload here
                'apiRequestId' => $apiRequestId,
                'messageIndex' => $messageIndex,
                'dispatcherJobConfig' => $interfaceConfig['dispatcherJobConfig'],
                'interfaceUdid' => $interfaceConfig['identity']['interface'],
            ],
            self::JOB_VERSION
        );
    }

    /**
     * Spawn a job.
     *
     * @param  string  $jobClass  namespace of the job class
     * @param  array  $overrides  optional overrides for the job keys
     */
    public function spawn(string $jobClass, array $overrides = []): void
    {

        if (Arr::get($overrides, 'isCorrectedResult', false) === true) {
            $overrides['data'] = $overrides;
            $overrides['queueType'] = 'ARTGEN';
            parent::spawn($jobClass, $overrides);

            return;
        }

        $nextJobClass = null;
        $nextJobs = $this->getNextJobsInWorkflow($overrides['orderId'], $jobClass);

        foreach ($nextJobs as $nextJob) {

            if (empty($nextJob['nextJobType'])) {
                continue;
            }
            if ($nextJob['nextJobType'] == 'callback') {
                $classObj = app($nextJob['class']);

                $results = call_user_func([$classObj, $nextJob['method']], $overrides, $this->model->getMasterOrder($overrides['orderId']));

                if ($results === null) {
                    // We didn't get any result back from the method, so continue to the next anchor in the loop
                    continue;
                }

                $nextJobClass = $results['job'];
                $overrides['queueType'] = $results['queue'];

                $data = array_exists('data', $results, []);
                $overridesData = array_exists('data', $overrides, []);

                $overrides['data'] = array_merge($overridesData, $data);
            } elseif ($nextJob['nextJobType'] == 'direct') {
                $nextJobClass = $nextJob['nextJob'];
                $overrides['queueType'] = $nextJob['queue'];
            }

            if (isset($nextJob['inputConfig']) && $nextJob['inputConfig']) {
                $inputConfig = Json::decode($nextJob['inputConfig']);
                $overridesData = array_exists('data', $overrides, []);
                $overrides['data'] = array_merge($overridesData, $inputConfig);
            }

            if (isset($nextJob['configuredWorkflowInput']) && $nextJob['configuredWorkflowInput']) {
                $configuredWorkflowInput = Json::decode($nextJob['configuredWorkflowInput']);
                $overridesData = array_exists('data', $overrides, []);
                $overrides['data'] = array_merge($overridesData, $configuredWorkflowInput);
            }

            // Spawn the new job
            parent::spawn($nextJobClass, $overrides);
        }

        // TODO Check if we should look to do an EOST Notification, check OrderJobSpawnerV2
    }

    public function getNextJobsInWorkflow(int $orderId, string $jobClass): mixed
    {
        $order = Order::find($orderId);
        if (!$order) {
            Log::error("Order not found for ID: " . $orderId);
            return [];
        }

        $workflowConfig = $order->productDefinition->workflow_config;

        if ($workflowConfig && array_key_exists($jobClass, $workflowConfig)) {
            $nextJobConfig = $workflowConfig[$jobClass];

            if (is_array($nextJobConfig) && array_key_exists(0, $nextJobConfig)) {
                return $nextJobConfig;
            } else {
                return [$nextJobConfig];
            }
        }

        return [];
    }
}

namespace App\Jobs;

use App\Exceptions\CustomJobException;
use App\Http\Services\V4\EventService;
use App\Models\Master\Gid\MasterGidCondition;

class OrderPublishJobV4 extends BaseWorkflowJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    protected $payload;

    protected $orderRepository;

    const DEFAULT_EVENT_CONTEXT = 'ORDER_WIP_PUBLISH';

    /**
     * Constructor to initialize job with payload
     *
     * @param  array  $payload  Job input data
     */
    public function __construct(array $payload)
    {
        $this->payload = $payload;
        $this->orderRepository = app(OrderRepository::class);
    }

    /**
     * Handle method that gets executed when job is processed
     */
    public function handle(): void
    {
        DB::beginTransaction();
        try {
            $payload = $this->payload;
            $orderId = (int) $this->job->getOrderId();
            $order = Order::with(['productDefinition', 'tests'])->findOrFail($orderId);

            $testCodeXref = $order->tests->first()->test_code_xref ?? null;
            if ($testCodeXref) {
                $testCodeXref = str_replace('%', '', $testCodeXref);
                $productDefinition = ProductDefinitions::where('test_code_xref', $testCodeXref)->first();
                if (!$productDefinition) {
                    throw new Exception("Product definition not found for test_code_xref: $testCodeXref");
                }
            } else {
                $productDefinition = $order->productDefinition;
            }

            if (!$productDefinition || !$productDefinition->config_data) {
                throw new Exception('Missing config_data in product definition.');
            }

            $endpointRouting = $productDefinition->config_data;
            $publishConfig = $endpointRouting['PublishToTechnical'] ?? [];

            if (empty($publishConfig)) {
                throw new Exception('Missing releaseToTechnical publish config.');
            }

            if (!empty($publishConfig['conditionsToSetOnSuccess'])) {
                $validations = [
                    'conditions' => [
                        'notPresent' => $publishConfig['conditionsToSetOnSuccess'],
                    ],
                ];
                $this->validateOrder($orderId, $validations);
            }

            $interfaceUdid = $publishConfig['interfaceUdid'];
            $eventUdid = $publishConfig['eventUdid'];
            $includes = $publishConfig['includes'] ?? [];
            $expectedResponseCode = $publishConfig['outboundInterfaceExpectedResponse'] ?? 201;
            $docType = $payload['docType'] ?? 'ORDER_EVENT_PUBLISH';

            $uri = app(QWorkerOrder::class)->getInterfaceEndpoint($interfaceUdid, $eventUdid);
            $authorization = app(Request::class)->getServiceAuthorization($interfaceUdid);
            $apiPayload = $this->assembleOrderPublishPayloadFromFilters($orderId, $includes, $docType, $order->productDefinition->test_code);

            if (isset($payload['triggeringEventId'])) {
                $apiPayload['triggeringEvent'] = $this->getTriggeringEventData($orderId, $payload['triggeringEventId'], $eventUdid);
            }

            $response = app(Request::class)->callOutboundApiRequest(
                $uri,
                [$authorization['headerKey'] . ': ' . $authorization['token']],
                $apiPayload,
                $authorization['systemUserApiCredentialId']
            );

            if ($response['responseStatusCode'] !== $expectedResponseCode) {
                throw new Exception('Unexpected response code: ' . $response['responseStatusCode']);
            }

            $sourceSystemUdid = $order->detail['meta']['sourceSystemUdid'] ?? null;
            $sourceSystem = SourceSystem::where('udid', $sourceSystemUdid)->first();

            if (!$sourceSystem) {
                throw new Exception("Invalid source system UDID: $sourceSystemUdid");
            }

            $eventService = app(EventService::class);
            $testCodes = $this->orderRepository->getTestCode($order);
            $orderEvent = $eventService->storeOrderEvent(
                $order,
                $eventUdid,
                $sourceSystem->udid,
                $response['apiRequestId'],
                $this->job->getLogId(),
                $response['apiRequestId'],
                $testCodes
            );

            foreach ($publishConfig['conditionsToSetOnSuccess'] ?? [] as $conditionCode) {
                $condition = MasterGidCondition::where('udid', $conditionCode)->first();

                if (!$condition) {
                    throw new Exception("Condition code not found: $conditionCode");
                }

                $order->conditions()->create([
                    'gid_condition_id' => $condition->gid_condition_id,
                ]);

                $order->conditionHistory()->create([
                    'gid_condition_id' => $condition->gid_condition_id,
                    'order_event_id' => $orderEvent->order_event_id,
                ]);
            }

            DB::commit();
        } catch (Exception $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT,
                null
            );
            $this->fail($customException);
        }
    }

    /**
     * Validates condition codes on the order
     *
     * @throws Exception
     */
    protected function validateOrder(int $orderId, array $validations): void
    {
        $problems = [];
        $conditions = $this->orderRepository->getOrderConditions($orderId);

        if (isset($validations['conditions']['present'])) {
            foreach ($validations['conditions']['present'] as $code) {
                if (!array_key_exists($code, $conditions)) {
                    $problems['conditionsThatShouldBePresent'][] = $code;
                }
            }
        }

        if (isset($validations['conditions']['notPresent'])) {
            foreach ($validations['conditions']['notPresent'] as $code) {
                if (array_key_exists($code, $conditions)) {
                    $problems['conditionsThatShouldNotBePresent'][] = $code;
                }
            }
        }

        if (!empty($problems)) {
            throw new Exception('Validation failed: ' . json_encode($problems));
        }
    }

    /**
     * Assemble outbound order payload based on includes
     */
    public function assembleOrderPublishPayloadFromFilters(

        string $testCode
    ): array {
        $clientData = $this->orderRepository->getClientData($orderId);
        $patientData = $this->orderRepository->getPatientData($orderId);
        if ($this->isProfileTest($testCode)) {
            return $this->getProfilePayload($orderId, $clientData, $patientData);
        }

        if ($testCode === 'T1050') {
            return $this->getMonitoringPayload($orderId, $clientData, $patientData);
        }

        $orderData = $this->orderRepository->getOrder($orderId);
        $accessionNumber = $this->extractAccessionNumber($orderData->adid);
        $createdDate = Carbon::parse($orderData->created)->utc()->format('Y-m-d\TH:i:s.u\Z');
        $sourceSystemUdid = $this->orderRepository->getsourceSystemUdid($orderId);

        return [
            'header' => [
                'createdOn' => Carbon::now()->toJSON(),
                'docType' => $docType,
                'docTypeVersion' => '1.0.0',
            ],
            'order' => [
                'orderAdid' => $orderData->adid,
                'AccessionNumber' => $accessionNumber,
                'RequisitionNumber' => '',
                'TestCode' => $orderData->test_code,
                'Testname' => $orderData->test_name,
                'AccessionDate' => $createdDate,
            ],
            'sourceSystem' => [
                'udid' => $sourceSystemUdid,

            'client' => $this->orderRepository->getClientData($orderId),
            'patient' => $this->orderRepository->getPatientData($orderId),

        ];
    }

class OrderResultDeliveryConfirmationJobV4 implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    const RESULT_DELIVERED_EVENT = 'RESULT_DELIVERED';

    const RESULT_CONFIRMED_CONDITION = 'RESULT_CONFIRMED'; // Must be a valid UDID in gid_conditions

    protected $api;

    public function __construct(
        protected array $payload,
    ) {
        $this->payload = $payload;
    }

    public function handle(): void
    {
        try {
            DB::beginTransaction();
            $this->api = app()->make(ApiModel::class);
            $jobId = (int) $this->job->getJobId();
            $apiRequestId = $this->payload['apiRequestId'];
            $apiPayload = $this->getMessage($apiRequestId);
            $messageIndex = $this->payload['messageIndex'] ?? 0;
            $dataIndex = $this->payload['dataIndex'] ?? 0;
            $message = $apiPayload['messages'][$messageIndex];
            $orderData = $message['payload']['data'][$dataIndex]['order'] ?? null;
            $orderAdid = $orderData['orderAdid'] ?? null;
            $order = Order::where('adid', $orderAdid)->first();
            $test = Test::where('order_id', $order->order_id)->first();
            if (in_array($test->test_code, ['T1030', 'T1040', 'T1050', 'PR1030'])) {
                $deliveryRepo = app(OrderResultDeliveryRepository::class);
                $resultReportPayloadId = $deliveryRepo->getResultReportPayloadIdFromOrderIDAndDigest([
                    'order_id' => $order->order_id,
                    'result_transform_adid' => $orderData['resultTransformAdid'] ?? '',
                ]);
                if (! $resultReportPayloadId) {
                    throw new RuntimeException(
                        'resultTransformAdid [%s] not found for orderAdid [%s]',
                        $orderData['resultTransformAdid'] ?? '',
                        $orderData['orderAdid'] ?? ''
                    );
                }
            }
            if (! $order) {
                Log::error('Order not found for result delivery confirmation', ['orderAdid' => $orderAdid]);
                throw new RuntimeException('Order not found for result delivery confirmatiom orderAdid ' . $orderAdid);
            }
            $sourceSystemUdid = $order->detail['meta']['sourceSystemUdid'] ?? null;
            $dataArray = $message['payload']['data'] ?? [];
            $allDeliveredTransforms = [];
            foreach ($dataArray as $dataItem) {
                $orderTransforms = $dataItem['order'] ?? [];
                $deliveredTransforms = $orderTransforms['deliveredTransforms'] ?? [];
                $allDeliveredTransforms[] = $deliveredTransforms;
            }
            $flattenedTransforms = array_merge(...$allDeliveredTransforms);
            $meta = json_encode($flattenedTransforms);
            $conditionId = $this->getConditionId('RDCF');
            $this->createCondition($order, $conditionId);
            $eventService = app(EventService::class);
            $orderEvent = $eventService->storeOrderEvent(
                $order,
                $message['payload']['event']['udid'],
                $sourceSystemUdid,
                $apiRequestId,
                $jobId,
                0,
                null,
                $meta
            );
            $this->createOrderConditionHistory($order, $orderEvent, $conditionId);
            DB::commit();
        } catch (Throwable $e) {
            // Rollback transaction if an error occurs
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            // Log the error and fail the job
            Log::error('OrderResultDeliveryConfirmationJobV4 failed', [
                'exception' => $e,
                'payload' => $this->payload,
            ]);

            throw $e;
        }
    }

    private function getConditionId(string $conditionCode): int
    {
        $condition = MasterGidCondition::where('udid', $conditionCode)->first();

        if (! $condition) {
            Log::error('Invalid condition UDID', ['udid' => $conditionCode]);
            throw new RuntimeException("Invalid condition UDID: $conditionCode");
        }

        return $condition->gid_condition_id;
    }

    private function createCondition(Order $order, int $conditionId): void
    {
        $order->conditions()->create([
            'gid_condition_id' => $conditionId,
        ]);
    }

    private function createOrderConditionHistory(Order $order, OrderEvent $event, int $conditionId): void
    {
        $order->conditionHistory()->create([
            'gid_condition_id' => $conditionId,
            'order_event_id' => $event->order_event_id,
        ]);
    }

    protected function getMessage(int $apiRequestId): array
    {
        return Json::decode($this->api->getInboundRequest($apiRequestId));
    }
}

class OrderResultTransformsJobV4 extends BaseWorkflowJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    protected $order = null;

    protected $orderEvent = null;

    protected $orderService = null;

    protected $eventService = null;

    protected $reportService = null;

    protected $attributes = [];

    /**
     * @var OrderRepository Handles database interaction with order conditions
     */
    protected OrderRepository $orderRepository;

    const DEFAULT_EVENT_CONTEXT = 'ORDER_RESULT_GENERATE';

    /**
     * Create a new job instance.
     */
    public function __construct(
        private $payload
    ) {

        $this->payload = $payload;
        $this->orderRepository = app(OrderRepository::class);
    }

    /**
     * Execute the job.
     */
    public function handle(): void
    {
        $orderId = $this->payload['orderId'];
        $this->orderService = app()->make(OrderService::class);
        $this->reportService = app()->make(ReportService::class);
        $this->eventService = app()->make(EventService::class);
        $this->order = $this->orderService->getOrderById($orderId);

        try {
            DB::beginTransaction();

            $this->initialize();

            if ($this->payload['report_id']) {
                $orderReport = OrderReport::where('report_id', $this->payload['report_id'])->first();
            } else {
                $orderReport = $this->createOrderReport();
            }

            $generatorConfigs = $this->getGeneratorConfig();
            foreach ($generatorConfigs as $generatorConfig) {
                $generatorClass = $generatorConfig['className'] ?? null;
                $generator = new $generatorClass;
                $generator->initialize($orderReport, $this->payload, $this->attributes);

                // TODO implement generate for each generator
                $message = $generator->generate();
                $generator->store([
                    'value' => $message,
                    'source_system_id' => $this->eventService->getSourceSystemByUdid($this->payload['sourceSystemUdid'])->source_system_id,
                ]);
            }
            $this->payload['report_id']=$orderReport->report_id;
            $orderJobSpawner = app(OrderJobSpawnerV4::class);
            $orderJobSpawner->spawn(get_class($this), [
                'orderId' => $this->order->order_id,
                'data' => $this->payload,
            ]);

            DB::commit();
        } catch (\Exception $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            // Use the CustomJobException
            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT,
                null
            );
            $this->fail($customException);
        }
    }

    private function createOrderReport(): OrderReport
    {
        $data = [
            'test_code' => $this->payload['test_code'],
            'report_status_udid' => $this->payload['rptStatusUdid'],
            'order_event_id' => $this->orderEvent->order_event_id ?? 0,
            'created_at' => now(),
            'reported_on' => $this->payload['reportedOn'] ?? now(),
            'meta' => $this->payload['meta'] ?? [],
        ];

        return $this->reportService->createOrderReport($this->order, $data);
    }

    private function initialize(): void
    {
        $this->validateOrder();

        $this->orderEvent = $this->createEvent();

        $this->createCondition();

        // TODO handle ARCO condition
    }

    protected function validateOrder(): void
    {
        if ($this->orderService->orderHasCondition($this->order, 'RLOCK')) {
            // throw new Exception('Order results are locked');
        }

        if ($this->orderService->orderHasCondition($this->order, 'WIPH')) {
            throw new \Exception(
                'Order updates not permitted. Order is currently on WIP HOLD, '
                .'contact operations to release for processing.'
            );
        }
    }

    /**
     * The product definition may contain report generator configurations.
     * "report_generators": [
     *      {
     *       "className": "App\\Jobs\\ReportGenerators\\HL7Generator",
     *      "skipOnConditionCodes": []
     *      },
     *      // more
     *  ]
     *
     * @TODO Handling an order with unrelated tests where the tests have different report generators.
     */
    private function getGeneratorConfig(): array
    {
        $productDefinition = $this->order->productDefinition;

        if (! $productDefinition && ! isset($productDefinition->report_generators)) {
            return [];
        }
        $generators = $productDefinition->report_generators;

        // Ensure LISAsciiGenerator runs first as ASCII report is needed for HL7 report
        $lisAsciiGenerators = array_filter($generators, function ($generator) {
            return str_contains($generator['className'] ?? '', 'LISAsciiGenerator');
        });

        $otherGenerators = array_filter($generators, function ($generator) {
            return ! str_contains($generator['className'] ?? '', 'LISAsciiGenerator');
        });

        return array_merge($lisAsciiGenerators, $otherGenerators);
    }

    private function createCondition(): void
    {
        if ($this->isFinalResult()) {
            $gidCondition = $this->orderService->getCondition('RLOCK');
            $gidConditionId = $gidCondition->gid_condition_id;

            $this->order->conditions()->create([
                'gid_condition_id' => $gidConditionId,
            ]);

            $this->createOrderConditionHistory($gidConditionId);
        }
    }

    private function createOrderConditionHistory(int $gidConditionId): void
    {
        $orderConditionHistory = $this->order->conditionHistory()->create([
            'gid_condition_id' => $gidConditionId,
            'order_event_id' => $this->orderEvent->order_event_id,
        ]);
        $this->attributes['orderConditionHistoryId'] = $orderConditionHistory->order_condition_history_id;
    }

    private function isFinalResult(): bool
    {
        return $this->payload['rptStatusUdid'] === 'FINAL';
    }

    private function createEvent(): OrderEvent
    {
        $inboundApiRequestId = $this->payload['apiRequestId'];
        $eventUdid = $this->payload['eventUdid'];
        $jobId = isset($this->payload['job_id']) ? (int) $this->payload['job_id'] : (int) $this->job->getLogId();
        $testCodes = $this->payload['test_code'];

        // TODO post_on should be saved, different from created_at
        return $this->eventService->storeOrderEvent(
            $this->order,
            $eventUdid,
            $this->payload['sourceSystemUdid'],
            $inboundApiRequestId,
            $jobId,
            0,
            $testCodes
        );
    }
}

    {

        $this->orderRepository = app()->make(OrderRepository::class);
        $this->sourceSystemRepository = app()->make(SourceSystemRepository::class);
        $this->testRepository = app()->make(TestRepository::class);
        $this->eventModel = app(SystemEventModel::class);
        $this->orderJobSpawner = app()->make(OrderJobSpawnerV4::class);
        $this->orderObj = app()->make(QWorkerOrder::class);
        $this->eventService = app()->make(EventService::class);
    }

    public function handle(): void
    {
        try {
            DB::beginTransaction();
            $payload = $this->payload;
            $orderId = $this->job->getOrderId();
            $jobId = $this->job->getLogId();

            $orderDetail = $this->orderRepository->getOrder($orderId);
            $testCodes = $this->orderRepository->getTestCode($orderDetail);
            $testCodes = ! empty($testCodes) ? explode(',', $testCodes)[0] : '';
            $this->eventService->storeOrderEvent(
                $orderDetail,
                $payload['eventUdid'],
                $orderDetail->detail['meta']['sourceSystemUdid'],
                $payload['apiRequestId'],
                $jobId,
                0,
                $testCodes
            );

            if ($this->shouldSendPatientUpdate($payload['newPatientData'])) {
                $this->sendPatientUpdateToExternalSystem($payload);
            }

            if ($this->shouldUpdateReport($payload)) {
                $reportStatus = $this->orderRepository->getReportStatusByOrderId($orderId);
                if ($reportStatus) {
                    $summary = $this->generateSummaryOfChanges($payload);
                    $orderedTest = $payload['testXref'] === 'PR1030'
                        ? ($reportStatus['udid'] === 'PRELIM' ? 'T1030' : 'T1040')
                        : $payload['testXref'];
                    $this->dispatchCorrectionEvent($orderId, $payload, $summary, $orderedTest);
                }
            }

            DB::commit();
        } catch (Exception $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT,
                null
            );
            $this->fail($customException);
        }
    }

    private function shouldSendPatientUpdate(array $newPatientData): bool
    {
        return isset($newPatientData['patientDOB']) ||
            isset($newPatientData['patientGivenName']) ||
            isset($newPatientData['patientSurname']);
    }

    private function shouldUpdateReport(array $payload): bool
    {
        return $this->shouldSendPatientUpdate($payload['newPatientData']) ||
            isset($payload['newPatientData']['patientInitials']) ||
            ! empty($payload['newClientData']);
    }

    private function sendPatientUpdateToExternalSystem(array $payload): void
    {
        $request = app()->make(Request::class);
        $uri = $this->orderObj->getInterfaceEndpoint($payload['interfaceUdid'], $payload['eventUdid']);

        $authorization = $request->getServiceAuthorization($payload['interfaceUdid']);

        $apiPayload = $this->assembleOrderPublishPayloadFromFilters($payload['orderId']);

        $request->callOutboundApiRequest(
            $uri,
            [
                $authorization['headerKey'].': '.$authorization['token'],
                'SL-API-Method: Update',
            ],
            $apiPayload,
            $authorization['systemUserApiCredentialId']
        );
    }

    private function generateSummaryOfChanges(array $payload): ?string
    {
        $existing = $payload['existingPatientAndClientData'];
        $newPatient = $payload['newPatientData'];
        $newClient = $payload['newClientData'];

        $changes = [];

        $fields = [
            'patientDob' => ['label' => 'date of birth', 'format' => true],
            'patientGivenName' => ['label' => 'given name'],
            'patientSurname' => ['label' => 'surname'],
            'patientInitials' => ['label' => 'middlename', 'fallback' => 'blank'],
            'clientSsidId' => ['label' => 'client id'],
            'clientSsidDesc' => ['label' => 'client name'],
        ];

        foreach ($fields as $key => $meta) {
            if (array_key_exists($key, $newPatient) || array_key_exists($key, $newClient)) {
                $old = $existing[$key] ?? ($meta['fallback'] ?? '');
                $new = $newPatient[$key] ?? $newClient[$key];
                if (! empty($meta['format'])) {
                    $old = date('m/d/Y', strtotime($old));
                    $new = date('m/d/Y', strtotime($new));
                }
                $changes[] = "{$meta['label']} from $old to $new";
            }
        }

        return ! empty($changes)
            ? "This report is amended to change the patient's ".implode(', ', $changes).'.'
            : null;
    }

class PostOrderInductionCheckJobV4 implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    protected $payload;

    protected $sourceSystemId = null;

    protected TestRepository $testRepository;

    protected OrderRepository $orderRepository;

    protected ReportRepository $reportRepository;

    protected OrderService $orderService;

    protected ProductRepository $productRepository;

    protected EventService $eventService;

    const DEFAULT_EVENT_CONTEXT = 'POST_ORDER_INDUCTION_CHECK';

    const EVENT_UDID = 'INB_ORD_POST_IND_CHECK';

    public function __construct($payload)
    {
        $this->payload = $payload;

        $this->testRepository = app(TestRepository::class);
        $this->orderRepository = app(OrderRepository::class);
        $this->reportRepository = app(ReportRepository::class);
        $this->orderService = app(OrderService::class);
        $this->productRepository = app(ProductRepository::class);
        $this->eventService = app(EventService::class);
    }

    public function handle(): void
    {
        try {
            DB::beginTransaction();

            $orderId = (int) $this->job->getOrderId();
            $inboundApiRequestId = $this->payload['apiRequestId'];
            $order = $this->orderRepository->getOrder($orderId);
            $orderAdid = $order->adid;
            $rootId = $order->root_id;
            $testCount = $order->tests->count();
            $failureReasons = [];

            foreach ($order->tests as $test) {
                $testCode = $this->productRepository->resolveTestCode($test);
                $productDef = $this->productRepository->getProductDefinition($testCode);

                if (! $productDef) {
                    $failureReasons[] = "Missing product definition for test_code: {$testCode}.";

                    continue;
                }

                if ($productDef->is_virtual) {
                    continue;
                }

                $failureReasons = array_merge(
                    $failureReasons,
                    $this->validateTestConditions($testCode, $orderAdid, $rootId, $testCount)
                );
            }
            $this->sourceSystemId = $this->orderRepository->getOrderSourceSystemId($orderId);
            $jobId = (int) $this->job->getLogId();
            $sourceSystem = SourceSystem::find($this->sourceSystemId);
            $sourceSystemUdid = $sourceSystem->udid;
            $testCodes = $this->orderRepository->getTestCode($order);
            $testCodes = ! empty($testCodes) ? explode(',', $testCodes)[0] : '';
            $event = $this->eventService->storeOrderEvent(
                $order,
                self::EVENT_UDID,
                $sourceSystemUdid,
                $inboundApiRequestId,
                $jobId,
                0,
                $testCodes
            );
            if (empty($failureReasons)) {
                $orderJobSpawner = app(OrderJobSpawnerV4::class);
                $orderJobSpawner->spawn(get_class($this), [
                    'orderId' => $orderId,
                    'data' => [
                        'apiRequestId' => $this->payload['apiRequestId'],
                    ],
                ]);
            } else {
                $this->eventService->setConditioCode($order, $event->order_event_id, 'WIPH');
                $reasons = implode(', ', $failureReasons);
                $this->orderRepository->addCommentsToEvents([
                    'order_event_id' => $event->order_event_id,
                    'systemComment' => "Post Order Induction Verification failed: $reasons",
                    'systemCommentTime' => Carbon::now(),
                ]);
            }

            DB::commit();
        } catch (\Exception $e) {
            if (DB::transactionLevel() > 0) {
                DB::rollBack();
            }

            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT
            );
            $this->fail($customException);
        }
    }

    /**
     * Validate test-specific data integrity rules.
     *
     * @return string[]
     */
    private function validateTestConditions(string $testCode, string $orderAdid, ?string $rootId, int $testCount): array
    {
        $reasons = [];

        switch ($testCode) {
            case 'PR1030':
                if ($testCount !== 3) {
                    $reasons[] = 'Test count must be 3 for test code PR1030.';
                }
                if ($rootId !== $orderAdid) {
                    $reasons[] = 'Root ID must match the Order ADID for Haystack Profile (PR1030).';
                }
                break;

            case 'T1050':
                if ($testCount !== 1) {
                    $reasons[] = 'Test count must be 1 for test code T1050.';
                    break;
                }

                // Determines if the root ID exists on any given V4 order
                $isRootExists = $this->orderRepository->checkRootIdExist($rootId);
                if (! $isRootExists) {
                    $reasons[] = 'Root ID does not exist for test code T1050.';
                    break;
                }

                // Determines if an order exists in the V4 scope
                if (Order::where('adid', $rootId)->exists()) {
                    $rootOrder = $this->orderRepository->getOrderByAdid($rootId);
                    $orderId = $rootOrder->order_id;
                    $orderTestCode = $rootOrder->productDefinition->test_code ?? null;

                    if ($orderTestCode !== 'PR1030') {
                        $reasons[] = 'Root ID does not belong to a PR1030 profile order for test code T1050.';
                    }

                    $finalReportGenerated = $this->reportRepository->orderFinalReportIsGenerated($orderId);
                    if (empty($finalReportGenerated)) {
                        $reasons[] = 'Final report is not generated yet for the linked profile order.';
                    }

                    break; // early return so this doesn't run twice
                }

                // Determines if an order exists in the V2 scope
                $rootOrder = app(V2OrderRepository::class)->get($rootId);
                $orderTestCode = $rootOrder->getTestCode();

                if ($orderTestCode !== 'PR1030') {
                    $reasons[] = 'Root ID does not belong to a PR1030 profile order for test code T1050.';
                }

                $finalReportGenerated = app(OrderResultRepository::class)->get($rootOrder);

                // Complicated conditional as $finalReportGenerated could be a collection, a model, or null
                if (
                    empty($finalReportGenerated) ||
                    (is_object($finalReportGenerated) && method_exists($finalReportGenerated, 'isEmpty') && $finalReportGenerated->isEmpty())
                ) {
                    $reasons[] = 'Final report is not generated yet for the linked profile order.';
                }

                break;

            case '14308':
            case '14309':
            case '14343':
                if ($testCount !== 1) {
                    $reasons[] = "Test count must be 1 for test code $testCode.";
                }
                break;
        }

        return $reasons;
    }

    private function generateAndSendReport(string $test, Order $order, SourceSystem $sourceSystem, $sourceSystemId, int $apiRequestId, array $payload)
    {

        $testPayload = $this->getTestPayload($this->orderId, $payload['orderedTest'], array_key_exists('reportStatus', $payload['data']) ? $payload['data']['reportStatus'] : null, $payload);

        $eventUdid = Arr::get($testPayload, 'EventUdid');

        /**
         * This condition is here because after the FINAL T1040 our T1030 is CORRECTED.
         * We're changing the orderedTest and the TestCode indexes to a T1040 because so that the PDF can generate
         * the T1040. Note by this point the T1040 TCOMT has been sent by STARLIMS.
         */
        $changedBaselineToPersonalization = false;
        $testPayload['OutboundHL7ReportStatus'] = $testPayload['OutboundHL7ReportStatus'] ?? null;
        if ($testPayload['OutboundHL7ReportStatus'] === 'CORRECTED' && $payload['orderedTest'] === 'T1030') {
            $payload['orderedTest'] = 'T1040';
            $testPayload['TestCode'] = 'T1040';
            $changedBaselineToPersonalization = true;
        }

        $payload['isCorrectedResult'] = false;
        if ($payload['orderedTest'] === 'T1030' && $eventUdid === 'OUT_ORD_RESAMME') {
            $payload['isCorrectedResult'] = true;
            $payload['reportStatus'] = 'CORRECTED';
        }

        $orderHeaderPayload = $this->buildOrderPayload($order, $sourceSystemId);
        $staticTextPayload = $this->buildStaticTextPayload();
        $sampleDatePayload = $this->buildSampleDatePayload($this->orderId, $testPayload['TestCode']);
        $mrdStreckBloodCollectionDate = Arr::get($sampleDatePayload, 'MRDCollectedDate', null);
        $ntePayload = $this->buildNtePayload(
            $order,
            $testPayload['TestCode'],
            $mrdStreckBloodCollectionDate,
            array_key_exists('summaryOfReportChanges', $payload['data']) ? $payload['data']['summaryOfReportChanges'] : '',
            $eventUdid
        );
        $reportPayload = array_merge(
            $ntePayload,
            $orderHeaderPayload,
            $staticTextPayload,
            $testPayload,
            $sampleDatePayload
        );

        // Optional clean-up if needed
        if (Arr::get($payload, 'reportAmendedWithTNPT')) {
            $this->makeFieldsNullForReporting($reportPayload);
        }

        // Calling outbound API
        $uri = $this->queueWorkerOrder->getInterfaceEndpoint(self::INTERFACE_UDID, self::EVENT_UDID);
        $request = app(Request::class);

        $authorization = $request->getServiceAuthorization(self::INTERFACE_UDID);
        $currentIsoDateString = Carbon::now()->toISOString();
        $signature = $this->make_signature($currentIsoDateString, $reportPayload);

        $headers = [
            $authorization['headerKey'].': '.$authorization['token'],
            'Report-Timestamp: '.$currentIsoDateString,
            'Report-Signature: '.$signature,
        ];

        $response = $request->callOutboundFileRequest(
            $uri,

    private function buildNtePayload(
        Order $order,
        string $testCode,
        ?string $mrdStreckBloodCollectionDate = null,
        ?string $summaryOfReportChanges = null,
        ?string $eventUdid = null
    ): array {
        $rootOrder = $this->orderRepository->getOrderByAdid($order->root_id);

        $nteValueMappings = ($testCode == self::MONITORING_TEST_CODE) ? array_merge($this->getMonitoringMappings(), $this->getProfileMappings($eventUdid)) : $this->getProfileMappings($eventUdid);

        $payload = $this->getMappingValues($order, $rootOrder, $nteValueMappings);

        // we are not getting ReportDate from StarLims in the mpdd
        $payload['ReportDate'] = Carbon::now('US/Central')->format('Y-m-d');
        if ($summaryOfReportChanges) {
            switch ($testCode) {
                case self::PERSONALIZATION_TEST_CODE:
                    $payload['PartialReportAmendedComment'] = $summaryOfReportChanges;
                    break;
                case self::BASELINE_TEST_CODE:
                    $payload['FinalReportAmendedComment'] = $summaryOfReportChanges;
                    break;
                case self::MONITORING_TEST_CODE:
                    $payload['ReportAmendedComment'] = $summaryOfReportChanges;
                    break;
            }
        }

        $testHistory = [];
        if (isset($this->payload['testCode'])) {
            $testCode = $this->payload['testCode'];
        }
        switch ($testCode) {
            case self::BASELINE_TEST_CODE:
                $collectionDate = Carbon::parse($mrdStreckBloodCollectionDate);
                $testHistoryTemp = [];
                $testHistoryTemp['CollectionDate'] = $collectionDate->format('m/d/y');
                $testHistoryTemp['MRDResult'] = $payload['MRDResult'];
                $testHistoryTemp['MMMPerMLPlasma'] = floatval($payload['MMMPerMLPlasma']);
                $testHistoryTemp['MMMDeviationLower'] = floatval($payload['MMMDeviationLower']);
                $testHistoryTemp['MMMDeviationUpper'] = floatval($payload['MMMDeviationUpper']);
                $testHistoryTemp['MutationsDetected'] = floatval($payload['MutationsDetected']);
                $testHistoryTemp['EvaluableMutations'] = floatval($payload['EvaluableMutations']);
                $testHistory[] = $testHistoryTemp;
                break;
            case self::MONITORING_TEST_CODE:
                $testHistory = self::buildMonitoringTestHistory($order, $rootOrder);
                break;
        }
        $payload['TestHistory'] = $testHistory;

        return $payload;
    }

    private function getMappingValues(Order $order, Order $rootOrder, array $nteValueMappings): array
    {
        $payload = [];
        foreach ($nteValueMappings as $key => $row) {

            $expendKey = explode('.', $key);
            $OrderTestcode = $expendKey[0];
            $identity = $expendKey[1];
            if (count($expendKey) == 4) {
                $upperIdentityKey = $expendKey[2];
                $identityKey = $expendKey[3];
            } else {
                $identityKey = $expendKey[2];
            }

            if ($OrderTestcode == 'T1040' || $OrderTestcode == 'T1030') {
                $CurrentOrder = $rootOrder;
            } else {
                $CurrentOrder = $order;
            }

            $orderTestDtails = $CurrentOrder->tests()->where('test_code', $OrderTestcode)->first();
            if ($identity == 'NTE' && ! empty($orderTestDtails->nte_details) && array_key_exists($identityKey, $orderTestDtails->nte_details)) {
                $payload[$row] = $orderTestDtails->nte_details[$identityKey]['value'];
            } elseif ($identity == 'OBX' && ! empty($orderTestDtails->detail[0]) && array_key_exists($identityKey, $orderTestDtails->detail[0])) {
                $payload[$row] = $orderTestDtails->detail[0][$identityKey]['resultValue'];
            } elseif ($identity == 'Order' && ! empty($CurrentOrder->detail[$upperIdentityKey]) && array_key_exists($identityKey, $CurrentOrder->detail[$upperIdentityKey])) {

                $identityKeyExpend = explode('-', $identityKey);
                if (count($identityKeyExpend) > 1) {
                    $payload[$row] = $CurrentOrder->detail[$upperIdentityKey][$identityKeyExpend[0]].' '.$CurrentOrder->detail[$upperIdentityKey][$identityKeyExpend[1]];
                } else {
                    $payload[$row] = $CurrentOrder->detail[$upperIdentityKey][$identityKeyExpend[0]];
                }
            } else {
                $payload[$row] = null;
            }
        }

        return $payload;
    }

    private function make_signature($currentTimestamp, $payload)
    {
        $serviceKey = getenv('REPORTING_SERVICE_KEY') ?: '';

        // these str_replaces are needed to match the json format in the python script in reporting service
        $jsonPayload = json_encode($payload);

        $payloadStamp = $currentTimestamp.PHP_EOL.$jsonPayload;

        // Convert the service key to UTF-8
        $key = mb_convert_encoding($serviceKey, 'UTF-8', mb_internal_encoding());

        return strtoupper(hash_hmac('sha256', $payloadStamp, $key));
    }

    /**
     * Get the test code of the ordered test
     *
     * @param  int  $apiRequestId  api request id
     * @param  string  $orderedTest  ordered test
     * @param  string|null  $reportStatus  report status
     * @return array ordered test data to be passed to reporting service
     */
    protected function getTestPayload(int $orderId, string $orderedTest, ?string $reportStatus, array $payload): array
    {

        $conditionCode = $payload['conditionCode'] ?? null;
        $testPayload = [];
        if ($reportStatus) {
            $testPayload['OutboundHL7ReportStatus'] = $reportStatus;

            if ($reportStatus === 'CORRECTED') {
                $testPayload['ReportStatus'] = $this->reportRepository->checkReportStatus($orderId, 'FINAL') ? 'FINAL' : 'PRELIM';
            }

            $testPayload['TestCode'] = $orderedTest;  // T1030/T1040/T1050
            $testPayload['EventUdid'] = $this->resultTypeToEventUdid($testPayload['ReportStatus']);

            return $testPayload;
        }

        $condition = $payload['conditionCode'] ?? null;
        if ($condition) {
            $testUdid = $orderedTest ?? null;
            $testPayload['OutboundHL7ReportStatus'] = $conditionCode == 'REPORTAMENDED' ? 'CORRECTED' : 'FINAL';
            $testPayload['ReportStatus'] = 'FINAL';
            if ($testUdid === 'T1030') {
                $testPayload['ReportStatus'] = 'PRELIM';
                $testPayload['OutboundHL7ReportStatus'] = 'PRELIM';
                if ($conditionCode == 'REPORTAMENDED') {
                    $testPayload['ReportStatus'] = $this->reportRepository->checkReportStatus($orderId, 'FINAL') ? 'FINAL' : 'PRELIM';
                    $testPayload['OutboundHL7ReportStatus'] = $this->reportRepository->checkReportStatus($orderId, 'FINAL') ? 'FINAL' : 'PRELIM';
                }
            }
            $testPayload['TestCode'] = $testUdid;
            $eventUdid = $this->resultTypeToEventUdid(isset($this->payload['resultType']) ? $this->payload['resultType'] : $testPayload['OutboundHL7ReportStatus']);
            if ($testUdid === 'T1030' && $conditionCode == 'REPORTAMENDED') {
                $eventUdid = $this->resultTypeToEventUdid('CORRECTED');
            }
            $testPayload['EventUdid'] = $eventUdid;
        }
        $tnpCases = ($conditionCode == 'TNPT' || $conditionCode == 'TNPW') ? $conditionCode : null;
        if ($tnpCases) {
            $tnpCode = $payload['tnp']['code'];
            $tnpMessage = $payload['tnp']['message'];
            $testPayload['TNPCode'] = $tnpCode;
            $testPayload['TNPMessage'] = $tnpMessage;
        }

        return $testPayload;
    }

    protected function resultTypeToEventUdid(string $resultType): string
    {
        switch ($resultType) {
            case self::RESULT_PRELIM:
                return self::EVENT_RESULT_PARTIAL;
            case self::RESULT_FINAL:
                return self::EVENT_RESULT_FINAL_STANDARD;
            case self::RESULT_ADDEND:
                return self::EVENT_RESULT_ADDEND;
            case self::RESULT_AMEND:
            case self::RESULT_CORRECTED:
                return self::EVENT_RESULT_AMENDED;
            case self::RESULT_RECLASS:
            case self::RESULT_DUPL:
                throw new Exception("No corresponding event exists for ResultType: $resultType");
            default:
                throw new Exception("Invalid ResultType: $resultType");
        }
    }

    private function sendReportPayloadToQdrs(array $reportPayload, string $report_id, Order $order, string $sourceSystemUdid): void
    {
        $currentTimestamp = Carbon::now()->format('Y-m-d\TH:i:s.u\Z');
        $request = app(Request::class);

        // don't want to include DateOfService in contentData array below
        $dateOfService = $reportPayload['DateOfService'];
        unset($reportPayload['DateOfService']);
        
        $prismOrderAdId = ['PrismOrderAdId' => $order->adid];

        // QDRS Mandated (Sample Collection Date).
        $dateCollected = $reportPayload['DatePlasmaCollected'];

        $tnpw = $this->conditionRepo->hasCondition($order->order_id, 'TNPW');
        $tnpp = $this->conditionRepo->hasCondition($order->order_id, 'TNPP');
        $tnpt = $this->conditionRepo->hasCondition($order->order_id, 'TNPT');
        $tnpwi = $this->conditionRepo->hasCondition($order->order_id, 'TNPWI');

        if ($tnpw || $tnpp || $tnpt || $tnpwi) {
            $dateCollected = $reportPayload['DatePlasmaReceived'];
        }

        $imagePayload = [
            'imageRepositoryRequest' => [
                'accessionCode' => $reportPayload['QLSAccessionNumber'],
                'altAccessionCd' => $reportPayload['labReferenceId'],  // altAccessionCd <-- labReferenceId <-- clientLabRefId
                'labCd' => $sourceSystemUdid,
                'patFirstName' => $reportPayload['patientGivenName'],
                'patLastName' => $reportPayload['patientSurname'],
                'patMiddleName' => isset($reportPayload['patientInitials']) ? $reportPayload['patientInitials'] : 'no-data',
                'clientNum' => $reportPayload['ClientID'],
                'dateOfBirth' => $reportPayload['DateOfBirth'],
                'dateOfService' => $dateOfService,
                'dateCollected' => $dateCollected,
                'reportingId' => $report_id,
                'imageContent' => [
                    'imageCode' => null,
                    'imageSeq' => 1,
                    'imageCaption' => null,
                    'contentData' => array_merge($prismOrderAdId, $reportPayload),
                    'compressInd' => null,
                    'contentProperty' => [
                        'classification' => 'MRD',
                        'category' => 'PRISM',
                        'width' => null,
                        'height' => null,
                        'id' => null,
                        'creationTimestamp' => $currentTimestamp,
                        'type' => 'JSON',
                    ],
                ],
            ],
        ];

        $uri = $this->queueWorkerOrder->getInterfaceEndpoint(self::QDRS_INTERFACE_UDID, self::QDRS_EVENT_UDID);
        $authorization = $request->getServiceAuthorization(self::QDRS_INTERFACE_UDID);

        $response = $request->callOutboundApiRequest(
            $uri,
            [$authorization['headerKey'].': '.$authorization['token']],
            $imagePayload,
            $authorization['systemUserApiCredentialId'],
            false,
            false,
            120
        );
        if ($response['responseStatusCode'] !== 200) {
            throw new Exception('Error Sending Report Payload To QDRS Image Repository');
        }
    }

namespace App\Jobs;

use App\Exceptions\CustomJobException;
use App\Http\Services\V4\EventService;
use App\Models\Master\SourceSystem;

use Exception;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Database\Eloquent\Casts\Json;

class RequestRootIdJobV4 extends BaseWorkflowJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    protected $payload;

    protected OrderRepository $orderRepository;

    protected EventService $eventService;

    protected $sourceSystemId;

    protected $orderId;

    protected $orderAdid;

    const DEFAULT_EVENT_CONTEXT = 'ORDER_ROOT_ID_REQUEST';

    const EVENT_UDID = 'OUT_ORD_REQ_ROOT_ID';

    const INTERFACE_UDID = 'ROOT_ID_SERVICE';

    const ROOT_ID_EVENT_CODE = 'QIINT';

    const MONITORING_TEST = 'T1050';

    const PERSONALIZATION_TEST = 'T1030';

    public function __construct($payload = [])
    {

        $this->payload = $payload;
        $this->orderRepository = app(OrderRepository::class);
        $this->eventService = app(EventService::class);
    }

    /**
     * requestRootIdSetJob handler
     */
    public function handle(): void
    {
        try {
            $payload = $this->payload;
            DB::beginTransaction();

            $this->orderId = (int) $this->job->getOrderId();

            $jobId = (int) $this->job->getLogId();

            $inboundApiRequestId = $payload['apiRequestId'];
            $order = $this->orderRepository->getOrder($this->orderId);
            if ($order->productDefinition->test_code === self::MONITORING_TEST) {
                $personalizationProductDefinition = ProductDefinitions::select(['test_structure', 'test_code_xref'])->where('test_code', self::PERSONALIZATION_TEST)->firstOrFail();
                $this->payload['analyteID'] = Arr::get($personalizationProductDefinition->test_structure, 'properties.items.elements.obx.elements.MRDROOTID.src');
                $this->payload['rootTestCode'] = $personalizationProductDefinition->test_code_xref;
            }
            $this->orderAdid = $order->adid;

            $this->sourceSystemId = $this->orderRepository->getOrderSourceSystemId($this->orderId);

            $sourceSystem = SourceSystem::find($this->sourceSystemId);

            $sourceSystemUdid = $sourceSystem->udid;

            $data = $this->requestRootId();
            $testCodes = $this->orderRepository->getTestCode($order);

            $event = $this->eventService->storeOrderEvent(
                $order,
                self::EVENT_UDID,
                $sourceSystemUdid,
                $inboundApiRequestId,
                $jobId,
                $data['apiRequestId'],
                $testCodes
            );

            if ($data['responseStatusCode'] === 201) {
                $this->eventService->setConditioCode($order, $event->order_event_id, self::ROOT_ID_EVENT_CODE);

                $this->orderRepository->addCommentsToEvents([
                    'order_event_id' => $event->order_event_id,
                    'systemComment' => 'Request Root ID requires Manual Intervention. Internal case ID is '.$data['apiRequestId'],
                    'systemCommentTime' => Carbon::now(),
                ]);
            } else {
                $responseData = Json::decode($data['response']);
                $this->orderRepository->storeRootId($this->orderId, $responseData['rootID'], $event->order_event_id);

                $orderJobSpawner = app(OrderJobSpawnerV4::class);
                $orderJobSpawner->spawn(get_class($this), [
                    'orderId' => $this->orderId,
                    'data' => [
                        'apiRequestId' => $inboundApiRequestId,
                    ],
                ]);
            }
            DB::commit();
        } catch (\Exception $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT,
                null
            );
            $this->fail($customException);
        }
    }

    /**
     * @throws Exception
     */
    protected function requestRootId(): array
    {
        $qWorkerOrder = app(QWorkerOrder::class);
        $uri = $qWorkerOrder->getInterfaceEndpoint(self::INTERFACE_UDID, self::EVENT_UDID);

        $request = app(Request::class);

        $authorization = $request->getServiceAuthorization(self::INTERFACE_UDID);

        $response = $request->callOutboundApiRequest(
            $uri,
            [$authorization['headerKey'].': '.$authorization['token']],
            $this->generateRootIdServiceBody(),
            $authorization['systemUserApiCredentialId']
        );

        if ($response['responseStatusCode'] !== 200 && $response['responseStatusCode'] !== 201) {
            throw new Exception('Error getting Root ID - Response: '.$response['response']);
        }

        $jobModel = app(JobModel::class);

        $jobId = (int) $this->job->getLogId();

        $jobModel->addOutboundRequests($jobId, [$response['apiRequestId']]);

        return $response;
    }

    protected function generateRootIdServiceBody(): array
    {
        $orderHeader = $this->orderRepository->getOrderHeader($this->orderId);

        $extractOrderAdid = explode('.', $this->orderAdid);

        $sourceSystemUdid = $extractOrderAdid[0];

        return [
            'options' => [
                'requestId' => 'prism-queue_job_log_id-'.$this->orderId,
                'minimumDegreeMatch' => '144',
                'DOSFrom' => Carbon::now()->subYears(20)->format('Y-m-d'),
                'DOSTo' => Carbon::now()->format('Y-m-d'),
                'patientProvidedRootId' => '',
            ],
            'accession' => [
                'orderAdid' => $this->orderAdid,
                'eid' => '',
                'rootTestCode' => isset($this->payload['rootTestCode']) ? '%'.$this->payload['rootTestCode'] : '',
                'analyteID' => (string) isset($this->payload['analyteID']) ? $this->payload['analyteID'] : '',
                'labSystemCode' => $sourceSystemUdid,
                'labAccessionCode' => $orderHeader['fields']['placerOrderXrefNumber'],  // placerOrderXrefNumber
                'caseToCreate' => 'MRD Root ID',
                'sfidAttribute' => 'Comments',
            ],
            'patient' => [
                'address' => [
                    'streetAddressLine' => isset($this->payload['address']['streetAddressLine']) ? $this->payload['address']['streetAddressLine'] : '',
                    'city' => isset($this->payload['address']['city']) ? $this->payload['address']['city'] : '',
                    'state' => isset($this->payload['address']['state']) ? $this->payload['address']['state'] : '',
                    'postalCode' => isset($this->payload['address']['postalCode']) ? $this->payload['address']['postalCode'] : '',
                    'country' => isset($this->payload['address']['country']) ? $this->payload['address']['country'] : '',
                ],
                'name' => [
                    'firstName' => $orderHeader['patient']['patientGivenName'],
                    'middleName' => '',
                    'lastName' => $orderHeader['patient']['patientSurname'],
                ],
                'dateOfBirth' => Carbon::parse($orderHeader['patient']['patientDOB'])->format('Y-m-d'),
                'sex' => $orderHeader['patient']['patientSex'],
            ],
        ];
    }
}

namespace App\Jobs;

use App\Exceptions\CustomJobException;
use App\Http\Services\V4\EventService;
use App\Models\Master\SourceSystem;

use Exception;
use Illuminate\Bus\Queueable;

class RootIdSetJobV4 extends BaseWorkflowJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    protected $payload;

    protected ?int $sourceSystemId = null;

    private OrderRepository $orderRepository;

    private EventService $eventService;

    const EVENT_UDID = 'OUT_ORD_REQ_ROOT_ID';

    const DEFAULT_EVENT_CONTEXT = 'HAYSTACK_ROOT_ID_SET';

    public function __construct($payload = [])
    {
        $this->payload = $payload;
        $this->orderRepository = app(OrderRepository::class);
        $this->eventService = app(EventService::class);
    }

    /**
     * HaystackRootIdSet jobV4 handler
     */
    public function handle(): void
    {
        try {
            DB::beginTransaction();
            $payload = $this->payload;
            $orderId = isset($this->payload['orderId']) ? (int) $this->payload['orderId'] : (int) $this->job->getOrderId();
            $inboundApiRequestId = $payload['apiRequestId'];

            $order = $this->orderRepository->getOrder($orderId);
            if (empty($order)) {
                throw new Exception("Order with ID {$orderId} not found.");
            }
            $this->sourceSystemId = $this->orderRepository->getOrderSourceSystemId($orderId);

            $sourceSystem = SourceSystem::find($this->sourceSystemId);
            $sourceSystemUdid = $sourceSystem->udid;
            $jobId = isset($this->payload['job_id']) ? (int) $this->payload['job_id'] : (int) $this->job->getLogId();
            $testCodes = $this->orderRepository->getTestCode($order);
            $testCodes = ! empty($testCodes) ? explode(',', $testCodes)[0] : '';
            $this->eventService->storeOrderEvent($order, self::EVENT_UDID, $sourceSystemUdid, $inboundApiRequestId, $jobId, 0, $testCodes);

            $orderAdid = $order->adid;

            if (! $orderAdid) {
                throw new Exception("Order Adid not found for Order ID {$orderId}, cannot set Root ID.");
            }

            $this->orderRepository->storeRootId($orderId, $orderAdid);
            $orderJobSpawner = app(OrderJobSpawnerV4::class);
            $orderJobSpawner->spawn(get_class($this), [
                'orderId' => $orderId,
                'data' => [
                    'apiRequestId' => $inboundApiRequestId,
                ],
            ]);

            DB::commit();
        } catch (\Exception $e) {
            if (DB::transactionLevel()) {
                DB::rollBack();
            }

            // Use the CustomJobException
            $customException = new CustomJobException(
                $e->getMessage(),
                $e->getCode(),
                $e,
                self::DEFAULT_EVENT_CONTEXT,
                null
            );
            $this->fail($customException);
        }
    }
